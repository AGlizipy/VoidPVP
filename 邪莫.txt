local function scanPlayerInventory(player)
	if not player then return "无结果" end
	local character = player.Character
	local backpack = player:FindFirstChild("Backpack")
	if character then
		local heldTool = character:FindFirstChildOfClass("Tool")
		if heldTool then return string.upper(heldTool.Name) end
	end
	if backpack then
		local knife = backpack:FindFirstChild("Knife")
		local gun = backpack:FindFirstChild("Gun")
		if knife then return "刀" end
		if gun then return "枪" end
		local anyTool = backpack:FindFirstChildOfClass("Tool")
		if anyTool then return string.upper(anyTool.Name) end
	end
	return "无结果"
end

local LOOK_ANGLE_THRESHOLD = 45
local STICKY_ANGLE_THRESHOLD = 80
local LOOK_DISTANCE_MAX     = 60
local STICKY_DISTANCE_MAX   = 90

local scannerRayParams = RaycastParams.new()
scannerRayParams.FilterType = Enum.RaycastFilterType.Exclude
scannerRayParams.IgnoreWater = true

local function findPlayerByLook()
	local camCF = Camera.CFrame
	local origin = camCF.Position
	local lookDir = camCF.LookVector
	local myChar = LocalPlayer.Character


	if stickyTarget then
		if stickyTarget.Parent and stickyTarget.Character then
			local head = stickyTarget.Character:FindFirstChild("Head")
			if head then
				local toTarget = head.Position - origin
				local dist = toTarget.Magnitude
				if dist <= STICKY_DISTANCE_MAX and dist > 0.1 then
					local dot = lookDir:Dot(toTarget.Unit)
					local angle = math.deg(math.acos(math.clamp(dot, -1, 1)))
					if angle < STICKY_ANGLE_THRESHOLD then
						scannerRayParams.FilterDescendantsInstances = {myChar, stickyTarget.Character}
						local result = workspace:Raycast(origin, toTarget, scannerRayParams)
						if not result then return stickyTarget, head end
					end
				end
			end
		end
		stickyTarget = nil
	end

	local bestPlayer, bestHead = nil, nil
	local bestAngle = LOOK_ANGLE_THRESHOLD
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			local head = player.Character:FindFirstChild("Head")
			if head then
				local toTarget = head.Position - origin
				local dist = toTarget.Magnitude
				if dist <= LOOK_DISTANCE_MAX and dist > 0.1 then
					local dot = lookDir:Dot(toTarget.Unit)
					local angle = math.deg(math.acos(math.clamp(dot, -1, 1)))
					if angle < bestAngle then
						scannerRayParams.FilterDescendantsInstances = {myChar, player.Character}
						local result = workspace:Raycast(origin, toTarget, scannerRayParams)
						if not result then
							bestAngle = angle
							bestPlayer = player
							bestHead = head
						end
					end
				end
			end
		end
	end
	if bestPlayer then stickyTarget = bestPlayer end
	return bestPlayer, bestHead
end

local function spawnGhostFromSnapshot(snapshot)
	local ghost = Instance.new("Model")
	ghost.Name = "残影分身"
	for _, info in ipairs(snapshot) do
		local part = Instance.new("Part")
		part.Name = info.Name
		part.Size = info.Size
		part.CFrame = info.CFrame
		part.Color = Color3.fromRGB(130, 200, 255)
		part.Material = Enum.Material.SmoothPlastic
		part.Transparency = 0
		part.Anchored = true
		part.CanCollide = false
		part.CanTouch = false
		part.CanQuery = false
		part.CastShadow = false
		part.Parent = ghost
	end
	ghost.Parent = workspace
	local hl = Instance.new("Highlight")
	hl.FillColor = Color3.fromRGB(130, 200, 255)
	hl.FillTransparency = 0.15
	hl.OutlineColor = Color3.fromRGB(200, 235, 255)
	hl.OutlineTransparency = 0
	hl.Adornee = ghost
	hl.Parent = ghost
	task.spawn(function()
		task.wait(0.15)
		for _, d in ipairs(ghost:GetChildren()) do
			if d:IsA("BasePart") then
				TweenService:Create(d, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Transparency = 1}):Play()
			end
		end
		TweenService:Create(hl, TweenInfo.new(0.5), {FillTransparency = 1, OutlineTransparency = 1}):Play()
		task.wait(0.55)
		if ghost and ghost.Parent then ghost:Destroy() end
	end)
end

local function dashOnce(moveVec)
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return false end

	local snapshot = {}
	for _, part in ipairs(char:GetChildren()) do
		if part:IsA("BasePart") and part.Name ~= "Head" then
			table.insert(snapshot, { Name = part.Name, Size = part.Size, CFrame = part.CFrame })
		end
	end

	hrp.CFrame = hrp.CFrame + moveVec * 12
	playSound(teleportSoundTemplate)

	local dh = Instance.new("Highlight")
	dh.FillColor = DASH_COLOR
	dh.FillTransparency = 0.2
	dh.OutlineColor = DASH_OUTLINE
	dh.OutlineTransparency = 0
	dh.Parent = char
	task.delay(0.25, function() if dh then dh:Destroy() end end)


	pcall(spawnGhostFromSnapshot, snapshot)
	return true
end

local function normalDash()
	if teleportCooldown then return end
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	teleportCooldown = true
	local dir = (hrp.CFrame.LookVector + hrp.CFrame.RightVector * dashSide).Unit
	dashOnce(dir)
	dashSide = -dashSide
	task.wait(0.5)
	teleportCooldown = false
end
local function defaultDash()
	if teleportCooldown then return end
	local char = LocalPlayer.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then return end
	teleportCooldown = true
	dashOnce(hrp.CFrame.LookVector)
	task.wait(0.5)
	teleportCooldown = false
end
local function doDash()
	if dashMode == "default" then defaultDash() else normalDash() end
end

local function updateDashIndicator()
	if dashIndicatorLabel then
		dashIndicatorLabel.Text = dashMode == "normal" and "冲刺：普通 [B]" or "冲刺：默认 [B]"
		dashIndicatorLabel.TextColor3 = DASH_OUTLINE
	end
end

local function setDashMode(mode)
	dashMode = mode
	updateDashIndicator()
end

local function toggleDashMode()
	if dashMode == "normal" then setDashMode("default") else setDashMode("normal") end
end

-- =====================================================
--  扫描器心跳
-- =====================================================
local raycastInterval = 0.08
local raycastTimer = 0

RunService.Heartbeat:Connect(function(deltaTime)
	if not scannerActive then return end
	raycastTimer = raycastTimer + deltaTime
	if raycastTimer < raycastInterval then return end
	raycastTimer = 0

	local foundPlayer, head = findPlayerByLook()
	if foundPlayer and head then
		if not hasTarget then
			hasTarget = true
			enableFilter()
			stopScannerSound()
			activeScannerSound = playSound(scannerSoundTemplate)
		end
		scannerGui.Parent = head
		scannerGui.Enabled = true
		characterHighlight.Adornee = foundPlayer.Character
		characterHighlight.Enabled = true
		local scanResult = scanPlayerInventory(foundPlayer)
		bottomLabel.Text = scanResult
		topLabel.Visible = (scanResult ~= "无结果")
	else
		if hasTarget then
			hasTarget = false
			fadeFilterOut(0.2)
			stopScannerSound()
		end
		scannerGui.Enabled = false
		characterHighlight.Enabled = false
		characterHighlight.Adornee = nil
	end
end)

local function toggleScanner()
	scannerActive = not scannerActive
	if scannerActive then
		hasTarget = false
		stickyTarget = nil
	else
		disableScanner()
	end
end

local function gunCFrame()
	return CFrame.new(gunConfig.posX, gunConfig.posY, gunConfig.posZ)
		* CFrame.Angles(math.rad(gunConfig.rotX), math.rad(gunConfig.rotY), math.rad(gunConfig.rotZ))
end

local function applyGunConfig()
	if not gunWelds then return end
	local s = gunConfig.scale
	for _, entry in ipairs(gunWelds) do
		if entry.weld and entry.weld.Parent then
			local baseCF = entry.baseCF
			local scaledCF = CFrame.new(baseCF.Position * s) * (baseCF - baseCF.Position)
			entry.weld.C0 = gunCFrame() * scaledCF
			if entry.part and entry.part.Parent and entry.baseSize then
				entry.part.Size = entry.baseSize * s
			end
		end
	end
	if muzzleWeld and muzzleWeld.Parent then
		local mo = GUN_STYLE.muzzleOffset or CFrame.new(0, 0, -1)
		local scaledMO = CFrame.new(mo.Position * s) * (mo - mo.Position)
		muzzleWeld.C0 = gunCFrame() * scaledMO
	end
end

local function positionModel(model, targetPos)
	local ok = pcall(function() model:PivotTo(CFrame.new(targetPos)) end)
	if not ok then
		local cf = model:GetBoundingBox()
		local offset = targetPos - cf.Position

			if part:IsA("BasePart") then part.CFrame = part.CFrame + offset end
		end
	end

local function findGround(pos, fromY)
	if fromY == nil then fromY = 400 end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignoreList()

local function rebuildGunBody()
	if not handle or not handle.Parent then return end
	if gunBodyFolder then gunBodyFolder:Destroy() end
	if muzzle and muzzle.Parent then muzzle:Destroy() end
	gunWelds = {}
	local parentTool = tool
	gunBodyFolder = Instance.new("Folder")
	gunBodyFolder.Name = "枪身"
	gunBodyFolder.Parent = parentTool
	local style = GUN_STYLE

	if style.type == "model" then
		local loaded, objects = pcall(function() return game:GetObjects(style.assetId) end)
		if loaded and objects and objects[1] then
			local source = objects[1]
			if not source:IsA("Model") then
				local wrapper = Instance.new("Model"); source.Parent = wrapper; source = wrapper
			end
			local sourcePivot = source:GetPivot()
			for _, obj in ipairs(source:GetDescendants()) do
				if obj:IsA("Script") or obj:IsA("LocalScript") or obj:IsA("ModuleScript") then obj:Destroy() end
			end
			for _, part in ipairs(source:GetDescendants()) do
				if part:IsA("BasePart") then
					local relative = sourcePivot:ToObjectSpace(part.CFrame)
					local baseSize = part.Size
					part.Anchored = false; part.CanCollide = false
					part.CanTouch = false; part.CanQuery = false; part.Massless = true
					part.Parent = gunBodyFolder
					local weld = Instance.new("Weld")
					weld.Name = "枪械部件焊接"
					weld.Part0 = handle; weld.Part1 = part
					weld.C0 = gunCFrame() * relative
					weld.C1 = CFrame.new()
					weld.Parent = handle
					table.insert(gunWelds, { weld = weld, baseCF = relative, part = part, baseSize = baseSize })
				end
			end
			source:Destroy()
		end
	end

	if #gunWelds == 0 then
		local fallback = Instance.new("Part")
		fallback.Name = "备用枪身"
		fallback.Size = Vector3.new(0.4, 0.4, 1.6)
		fallback.Color = PORTAL_COLOR
		fallback.Material = Enum.Material.Neon
		fallback.Anchored = false; fallback.CanCollide = false
		fallback.CanTouch = false; fallback.CanQuery = false; fallback.Massless = true
		fallback.Parent = gunBodyFolder
		local weld = Instance.new("Weld")
		weld.Name = "枪身焊接"
		weld.Part0 = handle; weld.Part1 = fallback
		weld.C0 = gunCFrame() * CFrame.new(0, 0, -0.8)
		weld.C1 = CFrame.new()
		weld.Parent = handle
		table.insert(gunWelds, { weld = weld, baseCF = CFrame.new(0, 0, -0.8), part = fallback, baseSize = fallback.Size })
	end

	muzzle = Instance.new("Part")
	muzzle.Name = "枪口"
	muzzle.Size = Vector3.new(0.2, 0.2, 0.2)
	muzzle.Transparency = 1
	muzzle.CanCollide = false
	muzzle.Massless = true
	muzzle.Parent = parentTool

	muzzleWeld = Instance.new("Weld")

	muzzleWeld.Part1 = muzzle
	muzzleWeld.C0 = gunCFrame() * (style.muzzleOffset or CFrame.new(0, 0, -1))
	muzzleWeld.C1 = CFrame.new()
	muzzleWeld.Parent = handle
end

local function removeGunFromPlayer(name)

		for _, v in ipairs(backpack:GetChildren()) do
			if v:IsA("Tool") and v.Name == name then v:Destroy() end
		end
	end
	if p.Character then
		for _, v in ipairs(p.Character:GetChildren()) do
			if v:IsA("Tool") and v.Name == name then v:Destroy() end
		end
	end
end

local function loadMap(assetId, name, center, scale)
	local ok, objs = pcall(function() return game:GetObjects(assetId) end)
	if not ok or not objs or not objs[1] then return nil end
	local model = objs[1]
	if not model:IsA("Model") then
		local w = Instance.new("Model"); model.Parent = w; model = w
	end
	model.Name = name; model.Parent = workspace
	positionModel(model, center)
	pcall(function() model:ScaleTo(scale) end)
	for _, part in ipairs(model:GetDescendants()) do
		if part:IsA("BasePart") then part.Anchored = true; part.CanCollide = true end
	end
	return model
end

local function getPortalCF(portalObj)
	if not portalObj or not portalObj.part or not portalObj.part.Parent then return nil end
	if portalObj.part:IsA("Model") then return portalObj.part:GetPivot() end
	return portalObj.part.CFrame
end

local function pointInside(portalObj, point)
	local cf = getPortalCF(portalObj)
	if not cf then return false end
	local rel = cf:PointToObjectSpace(point)
	if rel.Z > 2.2 or rel.Z < -1.8 then return false end
	return (rel.X*rel.X + rel.Y*rel.Y) <= 12.25
end

local function insidePortal(portal, root)
	if not portal or not portal.part or not portal.part.Parent then return false end
	local base = root.Position
	return pointInside(portal, base)
		or pointInside(portal, base - Vector3.new(0, 2.0, 0))
		or pointInside(portal, base + Vector3.new(0, 1.2, 0))
end

local function teleport(fromPortal, toPortal)
	local char = p.Character
	if not char or not toPortal or not toPortal.part or not toPortal.part.Parent then return end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum or hum.Health <= 0 then return end
	local now = os.clock()
	if now - lastTeleport < COOLDOWN or arrivedAt == fromPortal or isTeleporting then return end
	isTeleporting = true; lastTeleport = now
	if hum.SeatPart then hum.Sit = false end
	tSnd:Play()

	local enterPos, enterNorm = fromPortal.position, fromPortal.normal
	local exitPos, exitNorm = toPortal.position, toPortal.normal

	hrp.AssemblyLinearVelocity = Vector3.zero
	hrp.AssemblyAngularVelocity = Vector3.zero
	hrp.Anchored = true

	local enterLook = Vector3.new(-enterNorm.X, 0, -enterNorm.Z)
	enterLook = enterLook.Magnitude < 0.01 and hrp.CFrame.LookVector or enterLook.Unit
	local pullCF = CFrame.lookAt(enterPos - enterNorm*0.8, enterPos - enterNorm*0.8 + enterLook)
	local pullT = TweenService:Create(hrp, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {CFrame = pullCF})
	pullT:Play(); pullT.Completed:Wait()

	local launchVelocity, exitCF
	local flatLook = Vector3.new(hrp.CFrame.LookVector.X, 0, hrp.CFrame.LookVector.Z)
	if flatLook.Magnitude < 0.01 then flatLook = Vector3.new(0, 0, -1) else flatLook = flatLook.Unit end

	if exitNorm.Y > 0.7 then
		exitCF = CFrame.lookAt(exitPos + Vector3.new(0, 3.2, 0), exitPos + Vector3.new(0, 3.2, 0) + flatLook)
		launchVelocity = Vector3.new(0, 35, 0)
		hum:ChangeState(Enum.HumanoidStateType.Jumping)
	elseif exitNorm.Y < -0.7 then
		exitCF = CFrame.lookAt(exitPos - Vector3.new(0, 3.5, 0), exitPos - Vector3.new(0, 3.5, 0) + flatLook)
		launchVelocity = (flatLook * 15) + Vector3.new(0, -20, 0)
		hum:ChangeState(Enum.HumanoidStateType.Freefall)
	else
		local wallDir = Vector3.new(exitNorm.X, 0, exitNorm.Z)
		wallDir = wallDir.Magnitude < 0.01 and hrp.CFrame.LookVector or wallDir.Unit
		exitCF = CFrame.lookAt(exitPos + exitNorm * 3.0, exitPos + exitNorm * 3.0 + wallDir)
		launchVelocity = wallDir * 30
		hum:ChangeState(Enum.HumanoidStateType.Freefall)
	end

	hrp.CFrame = exitCF; hrp.Anchored = false
	hrp.AssemblyLinearVelocity = launchVelocity
	arrivedAt = toPortal; isTeleporting = false

	if toPortal.isReturnPortal then
		returnPosition = nil
		currentDestination = "普通"
		notify("返回", "你已返回地球！")
	end

	task.spawn(function()
		task.wait(0.4)
		while arrivedAt == toPortal do
			local c = p.Character
			local root = c and c:FindFirstChild("HumanoidRootPart")
			if not root or not toPortal.part or not toPortal.part.Parent or not insidePortal(toPortal, root) then
				arrivedAt = nil; break
			end
			task.wait(0.05)
		end
	end)
end

RunService.Heartbeat:Connect(function()
	local char = p.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local pos = root.Position
	local checks = {
		{spawnedChild,  CHILL_CENTER,  "休闲地图"},
		{spawnedGarage, GARAGE_CENTER, "车库"},
		{spawnedRetro,  RETRO_CENTER,  "复古地图"},
	}
	for _, c in ipairs(checks) do
		local map, center, name = c[1], c[2], c[3]
		if map and map.Parent and pos.Y < center.Y - 800 and pos.Y > center.Y - 4000 then
			root.CFrame = CFrame.new(center + Vector3.new(0, 20, 0))
			notify("保护机制", "你已被拉回 " .. name .. "！")
			return
		end
	end
end)

RunService.Heartbeat:Connect(function()
	if not portal1 or not portal2 or isTeleporting then return end
	local char = p.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	if insidePortal(portal1, root) then teleport(portal1, portal2)
	elseif insidePortal(portal2, root) then teleport(portal2, portal1) end
end)

local function createExitPortalInDimension(groundPos, hitNormal)
	local isFloorShot = math.abs(hitNormal.Y) > 0.7
	local exitPos, exitNorm
	if isFloorShot then
		exitPos = groundPos + Vector3.new(0, 0.05, 0)
		exitNorm = Vector3.new(0, 1, 0)
	else
		exitPos = groundPos + Vector3.new(0, EXIT_PORTAL_HEIGHT, 0)
		exitNorm = Vector3.new(0, 0, -1)
	end
	return createPortalPart(exitPos, exitNorm)
end

local function getAimPosition()
	local x, y = m.X, m.Y
	if x == 0 and y == 0 then
		local vp = Camera.ViewportSize
		x, y = vp.X / 2, vp.Y / 2
	end
	return x, y
end

local function shoot()
	local camera = workspace.CurrentCamera
	if not camera or not muzzle or not handle then return end
	local ax, ay = getAimPosition()
	local ray = camera:ScreenPointToRay(ax, ay)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignoreList()
	local result = workspace:Raycast(ray.Origin, ray.Direction * RANGE, params)
	local snd = handle:FindFirstChild("射击音效")
	if snd then snd:Play() end
	if not result then
		createBeam(muzzle.Position, ray.Origin + ray.Direction * RANGE, PORTAL_EMIT_COLOR)
		return
	end
	createBeam(muzzle.Position, result.Position, PORTAL_EMIT_COLOR)
	arrivedAt = nil

	local function resetBoth()
		if portal1 then destroyPortalPart(portal1) end
		if portal2 then destroyPortalPart(portal2) end
		portal1, portal2 = nil, nil
	end

	if currentDestination == "休闲" then
		resetBoth()
		portal1 = createPortalPart(result.Position, result.Normal)
		ensureChillExists()
		local gp = findGround(CHILL_CENTER, 400)
		portal2 = createExitPortalInDimension(gp, result.Normal)
		notify("通往休闲地图的传送门已开启！", "走进去！")
	elseif currentDestination == "车库" then
		resetBoth()
		portal1 = createPortalPart(result.Position, result.Normal)
		ensureGarageExists()
		local gp = findGround(GARAGE_CENTER, 0)
		portal2 = createExitPortalInDimension(gp, result.Normal)
		notify("通往车库的传送门已开启！", "你会出现在车库内部！")
	elseif currentDestination == "复古" then
		resetBoth()
		portal1 = createPortalPart(result.Position, result.Normal)
		ensureRetroExists()
		local gp = findGround(RETRO_CENTER, 400)
		portal2 = createExitPortalInDimension(gp, result.Normal)
		notify("通往复古地图的传送门已开启！", "走进去！")
	elseif currentDestination == "返回" then
		if not returnPosition then
			notify("提示", "返回点已丢失！")
			currentDestination = "普通"
			return
		end
		resetBoth()
		portal1 = createPortalPart(result.Position, result.Normal)
		local exitPos = returnPosition + Vector3.new(0, 3, 0)
		portal2 = createPortalPart(exitPos, Vector3.new(0, 0, -1), true)
		notify("返回地球的传送门已开启！", "走进去即可返回！")
	else
		if nextPortal == 1 then
			if portal1 then destroyPortalPart(portal1) end
			portal1 = createPortalPart(result.Position, result.Normal)
			nextPortal = 2
		else
			if portal2 then destroyPortalPart(portal2) end
			portal2 = createPortalPart(result.Position, result.Normal)
			nextPortal = 1
		end
	end
end

local function giveGun()
	cleanupPortals()
	removeGunFromPlayer(GUN_NAME)

	gunConfig.posX  = GUN_STYLE.pos.x
	gunConfig.posY  = GUN_STYLE.pos.y
	gunConfig.posZ  = GUN_STYLE.pos.z
	gunConfig.rotX  = GUN_STYLE.rot.x
	gunConfig.rotY  = GUN_STYLE.rot.y
	gunConfig.rotZ  = GUN_STYLE.rot.z
	gunConfig.scale = GUN_STYLE.scale
	refreshAllConfigRows()
	updateCodeBox()


	local okBuild, buildErr = pcall(function()
		local newTool, newHandle = buildEvilMortyGun()
		if not newTool or not newHandle then error("buildEvilMortyGun() 返回了 nil") end
		tool, handle = newTool, newHandle
	end)
	if not okBuild then
		warn("构建工具出错: " .. tostring(buildErr))
		return
	end
	pcall(rebuildGunBody)
	pcall(applyGunConfig)
	pcall(function() tool.Activated:Connect(function() shoot() end) end)

	local backpack = p:FindFirstChild("Backpack") or p:WaitForChild("Backpack", 10)
	if not backpack then return end
	tool.Parent = backpack

	if p.Character then
		local humanoid = p.Character:FindFirstChildOfClass("Humanoid")
		if humanoid and not p.Character:FindFirstChild(GUN_NAME) then
			pcall(function() humanoid:EquipTool(tool) end)
		end
	end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.R then toggleScanner() end
	if input.KeyCode == Enum.KeyCode.Q then doDash() end
	if input.KeyCode == Enum.KeyCode.B then toggleDashMode() end
	if input.KeyCode == Enum.KeyCode.Z then setModeNormal()
	elseif input.KeyCode == Enum.KeyCode.X then setModeChill()
	elseif input.KeyCode == Enum.KeyCode.C then setModeGarage()
	elseif input.KeyCode == Enum.KeyCode.E then setModeRetro()
	elseif input.KeyCode == Enum.KeyCode.V then setModeReturn()
	elseif input.KeyCode == Enum.KeyCode.G then
		if menuFrame then menuFrame.Visible = not menuFrame.Visible end
	elseif input.KeyCode == Enum.KeyCode.N then
		if configFrame then configFrame.Visible = not configFrame.Visible end
	end
end)

local savedPositions = {}
local editMode = false   -- false = 锁定（仅点击），true = 可拖动

local function createMobileControls()
	local playerGui = p:WaitForChild("PlayerGui")
	local old = playerGui:FindFirstChild("移动控制界面")
	if old then old:Destroy() end

	local gui = Instance.new("ScreenGui")
	gui.Name = "移动控制界面"
	gui.ResetOnSpawn = false
	gui.IgnoreGuiInset = true
	gui.DisplayOrder = 100
	gui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
	gui.Parent = playerGui

	-- ========= HUD 按钮 =========
	local function makeHUDButton(id, defaultPos, icon, caption, accentColor, onClick)
		local size = 52

		local container = Instance.new("Frame")
		container.Name = id
		container.Size = UDim2.new(0, size, 0, size + 14)
		container.BackgroundTransparency = 1
		container.Position = savedPositions[id] or defaultPos
		container.Parent = gui


		local btn = Instance.new("TextButton")
		btn.Size = UDim2.new(1, 0, 0, size)
		btn.BackgroundColor3 = Color3.fromRGB(10, 10, 10)
		btn.BackgroundTransparency = 0.35
		btn.BorderSizePixel = 0
		btn.Text = ""
		btn.AutoButtonColor = false
		btn.Active = true
		btn.Parent = container
		addCorner(btn, size / 2)

		local stroke = Instance.new("UIStroke")
		stroke.Color = accentColor
		stroke.Thickness = 2
		stroke.Transparency = 0.25
		stroke.Parent = btn

		local iconLbl = Instance.new("TextLabel")
		iconLbl.Size = UDim2.new(1, 0, 1, 0)
		iconLbl.BackgroundTransparency = 1
		iconLbl.Text = icon
		iconLbl.TextColor3 = accentColor
		iconLbl.TextSize = 26
		iconLbl.Font = Enum.Font.GothamBlack
		iconLbl.Parent = btn

		local cap = Instance.new("TextLabel")
		cap.Size = UDim2.new(1, 0, 0, 12)
		cap.Position = UDim2.new(0, 0, 1, 1)
		cap.BackgroundTransparency = 1
		cap.Text = caption
		cap.TextColor3 = Color3.fromRGB(230, 230, 230)
		cap.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
		cap.TextStrokeTransparency = 0.4
		cap.TextSize = 10
		cap.Font = Enum.Font.GothamBold
		cap.Parent = container

		-- ===== 逻辑：拖动或点击 =====
		local dragging = false
		local dragStart, startPos
		local moved = false
		local DRAG_THRESHOLD = 6

		btn.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				if editMode then
					dragging = true
					moved = false
					dragStart = input.Position
                                        					startPos = container.Position
				end
			end
		end)

		UserInputService.InputChanged:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseMovement
				or input.UserInputType == Enum.UserInputType.Touch then
				local delta = input.Position - dragStart
				if delta.Magnitude > DRAG_THRESHOLD then moved = true end
				container.Position = UDim2.new(
					startPos.X.Scale, startPos.X.Offset + delta.X,
					startPos.Y.Scale, startPos.Y.Offset + delta.Y
				)
			end
		end)

		UserInputService.InputEnded:Connect(function(input)
			if not dragging then return end
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then
				dragging = false
				if moved then
					savedPositions[id] = container.Position
				end
			end
		end)

		-- 普通点击（不在编辑模式时生效）
		btn.MouseButton1Click:Connect(function()
			if not editMode then
				onClick()
			end
		end)

		return container
	end

	-- 两个 HUD 按钮 — 左侧，偏下（避免挡住 HUD）
	makeHUDButton(
		"扫描按钮",
		UDim2.new(0, 20, 0.70, 0),
		"🔍", "扫描",
		Color3.fromRGB(255, 200, 40),
		function() toggleScanner() end
	)

	makeHUDButton(
		"冲刺按钮",
		UDim2.new(0, 20, 0.82, 0),
		"💨", "冲刺",
		Color3.fromRGB(0, 200, 255),
		function() doDash() end
	)

	-- ========= 编辑模式按钮 🔒/🔓 =========
	local lockBtn = Instance.new("TextButton")
	lockBtn.Name = "锁定按钮"
	lockBtn.Size = UDim2.new(0, 40, 0, 40)
	lockBtn.Position = UDim2.new(1, -56, 0, 16)
	lockBtn.BackgroundColor3 = Color3.fromRGB(10, 10, 10)
	lockBtn.BackgroundTransparency = 0.2
	lockBtn.BorderSizePixel = 0
	lockBtn.Text = editMode and "🔓" or "🔒"
	lockBtn.TextColor3 = editMode and Color3.fromRGB(120, 255, 120) or Color3.fromRGB(230, 230, 230)
	lockBtn.TextSize = 20
	lockBtn.Font = Enum.Font.GothamBlack
	lockBtn.AutoButtonColor = true
	lockBtn.Parent = gui
	addCorner(lockBtn, 20)

	local lockStroke = Instance.new("UIStroke")
	lockStroke.Color = editMode and Color3.fromRGB(120, 255, 120) or Color3.fromRGB(150, 150, 150)
	lockStroke.Thickness = 2
	lockStroke.Transparency = 0.3
	lockStroke.Parent = lockBtn

	lockBtn.MouseButton1Click:Connect(function()
		editMode = not editMode
		lockBtn.Text = editMode and "🔓" or "🔒"
		lockBtn.TextColor3 = editMode and Color3.fromRGB(120, 255, 120) or Color3.fromRGB(230, 230, 230)
		lockStroke.Color = editMode and Color3.fromRGB(120, 255, 120) or Color3.fromRGB(150, 150, 150)

		pcall(function() StarterGui:SetCore("SendNotification", {
			Title = editMode and "编辑模式已开启" or "编辑模式已关闭",
			Text = editMode and "可自由拖动按钮位置" or "按钮位置已锁定",
			Duration = 2
		}) end)
	end)
end
